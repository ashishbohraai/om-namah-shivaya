function getTTSPlugin() { if (window.Capacitor && window.Capacitor.Plugins) { return window.Capacitor.Plugins.TextToSpeech || null; } return null; } async function toggleSpeech(index) { if (currentlySpeakingIndex === index) { await stopSpeech(); return; } await stopSpeech(); // Clean punctuation/verse markers so the synthesizer reads smoothly const rawText = stotramData[index].sanskrit; const cleanText = rawText.replace(/[॥।\n]/g, ' '); const btn = document.getElementById(`tts-btn-${index}`); if (btn) { btn.innerText = '⏹ Stop'; btn.classList.add('speaking'); } currentlySpeakingIndex = index; const TTS = getTTSPlugin(); if (TTS) { try { await TTS.speak({ text: cleanText, lang: 'hi-IN', rate: 0.85, pitch: 1.0, volume: 1.0, category: 'ambient' }); } catch (err) { console.error("Native TTS Error:", err); fallbackWebSpeech(cleanText); } finally { resetTTSButtons(); currentlySpeakingIndex = null; } } else { fallbackWebSpeech(cleanText); } } function fallbackWebSpeech(text) { if (!('speechSynthesis' in window)) { alert("Text-to-speech engine is initializing or not ready."); resetTTSButtons(); currentlySpeakingIndex = null; return; } window.speechSynthesis.cancel(); const utterance = new SpeechSynthesisUtterance(text); utterance.lang = 'hi-IN'; utterance.rate = 0.85; utterance.onend = () => { resetTTSButtons(); currentlySpeakingIndex = null; }; utterance.onerror = () => { resetTTSButtons(); currentlySpeakingIndex = null; }; window.speechSynthesis.speak(utterance); } async function stopSpeech() { const TTS = getTTSPlugin(); if (TTS) { try { await TTS.stop(); } catch (e) {} } if ('speechSynthesis' in window) { window.speechSynthesis.cancel(); } resetTTSButtons(); currentlySpeakingIndex = null; } function resetTTSButtons() { document.querySelectorAll('.tts-btn').forEach(btn => { btn.innerText = '🔊 Listen'; btn.classList.remove('speaking'); }); }
